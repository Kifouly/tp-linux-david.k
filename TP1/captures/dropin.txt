# /etc/systemd/system/helios-collect.service
[Unit]
Description=Collecteur de telemetrie Helios
Documentation=https://wiki.helios.lan/collect
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=helios
Group=helios
Environment=HELIOS_INTERVAL=15
ExecStart=/usr/local/sbin/helios-collect.sh
Restart=on-failure
RestartSec=5s
TimeoutStopSec=20s
NoNewPrivileges=yes
PrivateTmp=yes
ProtectSystem=strict
ProtectHome=yes
ReadWritePaths=/var/lib/helios /var/log/helios
ProtectKernelTunables=yes
ProtectControlGroups=yes
RestrictSUIDSGID=yes
CapabilityBoundingSet=

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/helios-collect.service.d/override.conf
# /etc/systemd/system/helios-collect.service.d/override.conf
# Drop-in local : l'unit d'origine n'est jamais modifiee.

[Service]
# --- Reglages demandes par l'enonce (1.3) ---
Environment=HELIOS_INTERVAL=5
CPUQuota=20%
MemoryMax=96M
IOWeight=50
Nice=5

# --- Durcissement supplementaire (1.2 « notez le score, puis durcissez ») ---
# Le collecteur ne lit que /proc/loadavg et /proc/meminfo et ecrit dans
# /var/lib/helios : il n'a besoin ni de reseau, ni de peripheriques, ni
# d'acces au noyau.
PrivateDevices=yes
PrivateNetwork=yes
PrivateUsers=yes
ProtectKernelModules=yes
ProtectKernelLogs=yes
ProtectClock=yes
ProtectHostname=yes
RestrictNamespaces=yes
RestrictRealtime=yes
LockPersonality=yes
MemoryDenyWriteExecute=yes
RestrictAddressFamilies=AF_UNIX
SystemCallArchitectures=native
SystemCallFilter=@system-service
SystemCallFilter=~@privileged
SystemCallErrorNumber=EPERM
UMask=0027
# NB : pas de ProcSubset=pid, sinon /proc/loadavg et /proc/meminfo
# deviennent invisibles et le script plante.
