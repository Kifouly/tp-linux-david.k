$ cat /etc/zabbix/zabbix_agent2.d/helios-suivi.conf
# /etc/zabbix/zabbix_agent2.d/helios-suivi.conf - sondes du portail de suivi
# code HTTP de etat.json (200 attendu, 000 si le portail ne repond pas)
UserParameter=helios.suivi.http,/usr/bin/curl -s -o /dev/null -w '%{http_code}' --max-time 5 http://127.0.0.1:8080/etat.json
# age en secondes de etat.json (< 300 attendu, il est republie toutes les 2 min)
UserParameter=helios.suivi.age,echo $(( $(/usr/bin/date +%s) - $(/usr/bin/stat -c %Y /srv/helios/suivi/www/etat.json) ))
